% UDP: isolating congestion control from reassembly
% -------------------------------------------------------------------------------------
\begin{table}[htbp]
  \centering
  \begin{threeparttable}
    \caption{Per-track in-range loss on the highway. \texttt{BBox} is a single datagram and
             \texttt{PCloud} is 32, so comparing the two columns separates the cost of missing
             congestion control from the cost of all-or-nothing reassembly.}
    \label{tab:udp-loss}
    \begin{tabular}{@{}lrr@{}}
      \toprule
      Configuration & \texttt{BBox} (1 datagram) & \texttt{PCloud} (32 datagrams) \\
      \midrule
      MoQ / UDP  & \textbf{58.6\,\%} & 77.5\,\% \\
      MoQ / QUIC & 12.1\,\%          & 33.7\,\% \\
      MoQ / TCP  & 31.8\,\%          & 32.3\,\% \\
      \bottomrule
    \end{tabular}
    \begin{tablenotes}[flushleft]\footnotesize
      \item UDP loses 58.6\,\% of its \emph{single-datagram} messages, a gap that exists before
            any object is fragmented: its deficit is the absence of congestion control, not of
            reassembly robustness. Fragmentation adds a further ${\sim}1.8\times$ penalty
            (survival $41.4\,\% \rightarrow 22.5\,\%$), far less than an independence model
            would predict, because an object's datagrams are written back to back and are
            dropped together.
    \end{tablenotes}
  \end{threeparttable}
\end{table}

% RQ3: mapping onto the 3GPP eV2X catalogue
% -------------------------------------------------------------------------------------
\begin{table}[htbp]
  \centering
  \begin{threeparttable}
    \caption{The four enhanced-V2X categories of 3GPP TR 22.886 classified against the four
             properties a workload must have for MoQ's mechanisms to help.}
    \label{tab:ev2x}
    \begin{tabular}{@{}>{\RaggedRight}p{0.19\textwidth}>{\RaggedRight}p{0.28\textwidth}>{\RaggedRight}p{0.43\textwidth}@{}}
      \toprule
      Category & Workload shape & Verdict \\
      \midrule
      \textbf{Extended sensors} & Small CPM-like safety stream alongside bulk perception
        & \textbf{Measured here.} All four properties hold; this is the case this work answers. \\
      \addlinespace
      \textbf{Vehicle platooning} & Frequent small messages, little bulk to shed
        & Properties (1)--(2) barely apply: with no bulk track there is nothing to sacrifice.
          The BDP-sized window still helps; the delivery timeout has little to act on.
          \emph{Inferred, untested.} \\
      \addlinespace
      \textbf{Advanced driving} & Coordination messages alongside sensor sharing
        & Plausibly satisfies all four, depending on the sensor payload.
          \emph{Inferred, untested.} \\
      \addlinespace
      \textbf{Remote driving} & Large uplink video \emph{is} the latency-critical stream
        & \textbf{Violates property (1).} There is no bulk track to sacrifice, so shedding would
          attack the critical stream itself. Predicted unsuitable on structural grounds. \\
      \bottomrule
    \end{tabular}
    \begin{tablenotes}[flushleft]\footnotesize
      \item Properties: (1) the latency-critical stream is small relative to competing traffic;
            (2) competing traffic is loss-tolerant and independently interpretable;
            (3) deadlines are known a priori and static; (4) the transport queue can be bounded
            near the bandwidth-delay product.
      \item Only the first row is measured. The others are classified by matching their
            TS 22.186 requirements against these properties, which is reasoning rather than
            evidence.
    \end{tablenotes}
  \end{threeparttable}
\end{table}

  \begin{threeparttable}
    \caption{Defects found in third-party simulators. Each silently corrupts results, and each
             was found by investigating an implausible number rather than by a visible failure.}
    \label{tab:upstream-bugs}
    \begin{tabular}{@{}>{\RaggedRight}p{0.13\textwidth}>{\RaggedRight}p{0.38\textwidth}>{\RaggedRight}p{0.39\textwidth}@{}}
      \toprule
      Where & Defect & Effect \\
      \midrule
      \textbf{Simu5G} & \texttt{computeRuralMacro} passed the carrier frequency in \textbf{Hz}
        into a TR 38.901 term specified in \textbf{GHz}
        & A constant \textbf{+180\,dB} of phantom path loss; \texttt{RURAL\_MACROCELL} returned
          252--281\,dB and no radio link survived. Cross-checked against ns-3's independent
          implementation. \textbf{This blocked the entire highway scenario.} \\
      \addlinespace
      \textbf{INET QUIC} & \textsc{reset\_stream} released stream-level but not
        \textbf{connection-level} flow-control credit for abandoned bytes (RFC 9000 §4.5)
        & The connection window leaked shut on every reset, deadlocking any reset-heavy run. \\
      \addlinespace
      \textbf{INET QUIC} & Send-side flow control was a running byte counter that
        \textbf{decremented on loss} and never re-charged retransmissions (RFC 9000 §4.1 charges
        the \emph{highest offset sent})
        & The sender credited itself window it had already spent, overran the peer's limit, and
          the connection aborted with \textsc{flow\_control\_error} at small windows. \\
      \bottomrule
    \end{tabular}
  \end{threeparttable}
